cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Programmer board")
include(../pico_sdk_import.cmake)
project(atxmega192a3u_programmer C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_RP2350)
    message(FATAL_ERROR "This PDI transport requires RP2350 (Pico 2)")
endif()
add_executable(atxmega192a3u_programmer
    ../src/programmer_main.c ../src/pdi_transport.c ../src/xmega_nvm.c)
target_include_directories(atxmega192a3u_programmer PRIVATE ../src)
target_compile_definitions(atxmega192a3u_programmer PRIVATE PICO_STACK_SIZE=0x1000)
set_source_files_properties(../src/programmer_main.c ../src/pdi_transport.c ../src/xmega_nvm.c
    PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror;-fstack-usage")
pico_generate_pio_header(atxmega192a3u_programmer ${CMAKE_CURRENT_LIST_DIR}/../src/avr_pdi.pio)
target_link_libraries(atxmega192a3u_programmer pico_stdlib hardware_pio hardware_clocks)
pico_enable_stdio_uart(atxmega192a3u_programmer 0)
pico_enable_stdio_usb(atxmega192a3u_programmer 1)
pico_add_extra_outputs(atxmega192a3u_programmer)
